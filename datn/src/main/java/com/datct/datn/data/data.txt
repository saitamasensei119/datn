-- =========================
-- USERS
-- =========================
CREATE TABLE users (
    id BIGSERIAL PRIMARY KEY,
    full_name VARCHAR(255) NOT NULL,
    email VARCHAR(255) UNIQUE NOT NULL,
    password VARCHAR(255) NOT NULL,
    role VARCHAR(50) NOT NULL,

    enabled BOOLEAN NOT NULL DEFAULT TRUE,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- =========================
-- DEPARTMENTS
-- =========================
CREATE TABLE departments (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL UNIQUE
);

-- =========================
-- STUDENTS
-- =========================
CREATE TABLE students (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT UNIQUE NOT NULL,
    student_code VARCHAR(50) UNIQUE NOT NULL,
    department_id BIGINT,

    status VARCHAR(30) NOT NULL DEFAULT 'ACTIVE',
    --ACTIVE,        đang học
    --SUSPENDED,     tạm ngừng học
    --DROPPED_OUT,   thôi học
    --GRADUATED      đã tốt nghiệp
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_students_user
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_students_department
        FOREIGN KEY (department_id)
        REFERENCES departments(id)
);

-- =========================
-- LECTURERS
-- =========================
CREATE TABLE lecturers (
    id BIGSERIAL PRIMARY KEY,
    user_id BIGINT UNIQUE NOT NULL,
    lecturer_code VARCHAR(50) UNIQUE NOT NULL,
    department_id BIGINT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    status VARCHAR(30) NOT NULL DEFAULT 'ACTIVE'
    --ACTIVE    : đang công tác
    --ON_LEAVE  : nghỉ phép / tạm nghỉ
    --RETIRED   : nghỉ hưu
    --RESIGNED  : nghỉ việc

    CONSTRAINT fk_lecturers_user
        FOREIGN KEY (user_id)
        REFERENCES users(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_lecturers_department
        FOREIGN KEY (department_id)
        REFERENCES departments(id)
);

-- =========================
-- SEMESTERS
-- =========================
CREATE TABLE semesters (
    id BIGSERIAL PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    start_date DATE,
    end_date DATE
);

-- =========================
-- SUBJECTS
-- =========================
CREATE TABLE subjects (
    id BIGSERIAL PRIMARY KEY,
    subject_code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(255) NOT NULL,

    english_name VARCHAR(255),
    subject_type VARCHAR(30) NOT NULL,
    -- BT, ĐA, ĐATN, ĐATNKS, LT, LT+BT, TH, TN, TT, TTKS, TTKT, TTTN
    lab_requirement VARCHAR(20),
    program_code VARCHAR(50),
    note TEXT,

    management_code VARCHAR(30) NOT NULL,
    -- ELITECH,CT_CHUAN,KSCSDT,SIE,VB2CQ_VLVH,VLVH

    credits INT NOT NULL,
    department_id BIGINT,

    CONSTRAINT fk_subjects_department
        FOREIGN KEY (department_id)
        REFERENCES departments(id)
);

-- =========================
-- COURSES (LỚP HỌC PHẦN)
-- =========================
CREATE TABLE courses (
    id BIGSERIAL PRIMARY KEY,
    subject_id BIGINT NOT NULL,
    lecturer_id BIGINT,
    semester_id BIGINT NOT NULL,
    course_code VARCHAR(50) UNIQUE NOT NULL,
    max_students INT NOT NULL,
    status VARCHAR(20) NOT NULL DEFAULT 'PLANNED',
    PLANNED,      -- chuẩn bị mở
    OPEN,         -- đang cho đăng ký
    IN_PROGRESS,  -- đang học
    COMPLETED,    -- kết thúc
    CANCELLED     -- hủy

    opening_batch VARCHAR(2) NOT NULL,

    attached_course_code VARCHAR(50) NOT NULL,
    note TEXT NOT NULL,

    CONSTRAINT fk_courses_subject
        FOREIGN KEY (subject_id)
        REFERENCES subjects(id),

    CONSTRAINT fk_courses_lecturer
        FOREIGN KEY (lecturer_id)
        REFERENCES lecturers(id),

    CONSTRAINT fk_courses_semester
        FOREIGN KEY (semester_id)
        REFERENCES semesters(id)
);

-- =========================
-- ENROLLMENTS (ĐĂNG KÝ MÔN)
-- =========================
CREATE TABLE enrollments (
    id BIGSERIAL PRIMARY KEY,
    student_id BIGINT NOT NULL,
    course_id BIGINT NOT NULL,
    enrolled_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_enrollments_student
        FOREIGN KEY (student_id)
        REFERENCES students(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_enrollments_course
        FOREIGN KEY (course_id)
        REFERENCES courses(id)
        ON DELETE CASCADE,

    CONSTRAINT unique_student_course
        UNIQUE(student_id, course_id)
);

-- =========================
-- ROOMS
-- =========================
CREATE TABLE rooms (
    id BIGSERIAL PRIMARY KEY,
    room_name VARCHAR(100) UNIQUE NOT NULL,
    capacity INT NOT NULL
);

-- =========================
-- TIMESLOTS
-- =========================
CREATE TABLE timeslots (
    id BIGSERIAL PRIMARY KEY,
    day_of_week INT NOT NULL, -- 1=Monday ... 7=Sunday
    start_time TIME NOT NULL,
    end_time TIME NOT NULL
);

-- =========================
-- SCHEDULES
-- =========================
CREATE TABLE schedules (
    id BIGSERIAL PRIMARY KEY,
    course_id BIGINT NOT NULL,
    room_id BIGINT NOT NULL,
    timeslot_id BIGINT NOT NULL,

    CONSTRAINT fk_schedules_course
        FOREIGN KEY (course_id)
        REFERENCES courses(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_schedules_room
        FOREIGN KEY (room_id)
        REFERENCES rooms(id),

    CONSTRAINT fk_schedules_timeslot
        FOREIGN KEY (timeslot_id)
        REFERENCES timeslots(id),

    CONSTRAINT unique_room_timeslot
        UNIQUE(room_id, timeslot_id),

    CONSTRAINT unique_course_timeslot
        UNIQUE(course_id, timeslot_id)
);

-- =========================
-- GRADES
-- =========================
CREATE TABLE grades (
    id BIGSERIAL PRIMARY KEY,
    enrollment_id BIGINT UNIQUE NOT NULL,
    midterm_score NUMERIC(4,2),
    final_score NUMERIC(4,2),
    total_score NUMERIC(4,2),

    CONSTRAINT fk_grades_enrollment
        FOREIGN KEY (enrollment_id)
        REFERENCES enrollments(id)
        ON DELETE CASCADE
);

-- =========================
-- POSTS
-- =========================
CREATE TABLE posts (
    id BIGSERIAL PRIMARY KEY,
    title VARCHAR(255) NOT NULL,
    content TEXT NOT NULL,
    author_id BIGINT,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_posts_author
        FOREIGN KEY (author_id)
        REFERENCES users(id)
        ON DELETE SET NULL
);

-- =========================
-- grade components
-- =========================
CREATE TABLE grade_components (
    id BIGSERIAL PRIMARY KEY,
    course_id BIGINT NOT NULL,

    midterm_weight NUMERIC(4,2) NOT NULL,

    CONSTRAINT fk_gc_course
        FOREIGN KEY (course_id)
        REFERENCES courses(id)
);

-- =========================
-- grade submissions
-- =========================
CREATE TABLE grade_submissions (
    id BIGSERIAL PRIMARY KEY,

    course_id BIGINT NOT NULL,

    grade_type VARCHAR(20) NOT NULL,
    -- MIDTERM, FINAL

    status VARCHAR(30) NOT NULL,
    -- NOT_SUBMITTED  vẫn sửa điểm được
    -- SUBMITTED      không sửa điểm được

    submitted_at TIMESTAMP,
    locked_at TIMESTAMP,

    CONSTRAINT fk_gs_course
        FOREIGN KEY (course_id)
        REFERENCES courses(id)
);

-- =========================
-- subject conditions
-- =========================
CREATE TABLE subject_conditions (
    id BIGSERIAL PRIMARY KEY,

    subject_id BIGINT NOT NULL,
    required_subject_id BIGINT NOT NULL,

    condition_type VARCHAR(50) NOT NULL,
    -- 'PREREQUISITE',
    -- 'PRE_STUDY',
    -- 'COREQUISITE',
    -- 'EQUIVALENT'

    CONSTRAINT fk_subject_condition_subject
        FOREIGN KEY (subject_id)
        REFERENCES subjects(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_subject_condition_required_subject
        FOREIGN KEY (required_subject_id)
        REFERENCES subjects(id)
        ON DELETE CASCADE,

    CONSTRAINT uq_subject_condition
        UNIQUE (
            subject_id,
            required_subject_id,
            condition_type
        ),
    CONSTRAINT chk_subject_condition_self
        CHECK (subject_id <> required_subject_id)
);

-- =========================
-- student subject results
-- =========================
CREATE TABLE student_subject_results (
    id BIGSERIAL PRIMARY KEY,

    student_id BIGINT NOT NULL,
    subject_id BIGINT NOT NULL,

    best_attempt_enrollment_id BIGINT,

    final_score DECIMAL(4,2) NOT NULL,
    letter_grade VARCHAR(2),
    grade_point DECIMAL(3,2),

    is_passed BOOLEAN NOT NULL,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_result_student
        FOREIGN KEY (student_id)
        REFERENCES students(id),

    CONSTRAINT fk_result_subject
        FOREIGN KEY (subject_id)
        REFERENCES subjects(id),

    CONSTRAINT fk_result_best_attempt_enrollment
        FOREIGN KEY (best_attempt_enrollment_id)
        REFERENCES enrollments(id),

    CONSTRAINT uq_result
        UNIQUE(student_id, subject_id)
);

-- =========================
-- attendance sessions
-- =========================
CREATE TABLE attendance_sessions (
    id BIGSERIAL PRIMARY KEY,

    course_id BIGINT NOT NULL,

    session_date DATE NOT NULL,

    note TEXT,

    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,

    CONSTRAINT fk_attendance_session_course
        FOREIGN KEY (course_id)
        REFERENCES courses(id)
        ON DELETE CASCADE
);
-- =========================
-- attendance records
-- =========================
CREATE TABLE attendance_records (
    id BIGSERIAL PRIMARY KEY,

    attendance_session_id BIGINT NOT NULL,

    student_id BIGINT NOT NULL,

    status VARCHAR(20) NOT NULL,
    -- PRESENT
    -- ABSENT
    -- LATE
    -- EXCUSED

    note TEXT,

    CONSTRAINT fk_attendance_record_session
        FOREIGN KEY (attendance_session_id)
        REFERENCES attendance_sessions(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_attendance_record_student
        FOREIGN KEY (student_id)
        REFERENCES students(id),

    CONSTRAINT uq_attendance_record
        UNIQUE(attendance_session_id, student_id)
);

-- =========================
-- class schedules
-- =========================
CREATE TABLE class_schedules (
    id BIGSERIAL PRIMARY KEY,

    course_id BIGINT NOT NULL,

    session_number INT NOT NULL,

    day_of_week INT NOT NULL,

    start_period INT NOT NULL,

    end_period INT NOT NULL,

    shift VARCHAR(20), --kíp

    time_string VARCHAR(50),

    week_pattern VARCHAR(100),

    room_id BIGINT,

    CONSTRAINT fk_class_schedule_course
        FOREIGN KEY (course_id)
        REFERENCES courses(id)
        ON DELETE CASCADE,

    CONSTRAINT fk_class_schedule_room
        FOREIGN KEY (room_id)
        REFERENCES rooms(id)
);
